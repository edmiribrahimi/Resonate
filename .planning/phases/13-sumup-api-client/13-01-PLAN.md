```typescript
import SumUp, { APIError } from "@sumup/sdk";

// SDK singleton -- reused across all calls, do NOT instantiate per-request
const sumup = new SumUp({
  apiKey: process.env.SUMUP_API_KEY!,
});

// Export singleton for direct SDK access in future phases (14-18)
export { sumup };

export async function createCheckout(params: {
  amount: number;
  currency: string;
  description: string;
  checkoutReference: string;
  returnUrl: string;
}) {
  try {
    const checkout = await sumup.checkouts.create({
      amount: params.amount,
      currency: params.currency as "EUR",
      merchant_code: process.env.SUMUP_MERCHANT_CODE!,
      checkout_reference: params.checkoutReference,
      description: params.description,
      return_url: params.returnUrl,
    });

    return checkout as { id: string; status: string; checkout_reference: string };
  } catch (error) {
    if (error instanceof APIError) {
      throw new Error(
        `SumUp checkout creation failed: ${JSON.stringify(error.error)}`
      );
    }
    throw error;
  }
}

export async function getCheckout(checkoutId: string) {
  try {
    const checkout = await sumup.checkouts.get(checkoutId);

    return checkout as {
      id: string;
      status: "PENDING" | "PAID" | "FAILED" | "EXPIRED";
      checkout_reference: string;
      amount: number;
      currency: string;
      transactions: Array<{
        id: string;
        transaction_code: string;
        status: string;
      }>;
    };
  } catch (error) {
    if (error instanceof APIError) {
      throw new Error(`SumUp checkout retrieval failed: ${error.status}`);
    }
    throw error;
  }
}

export async function refundTransaction(transactionCode: string, amount?: number) {
  try {
    await sumup.transactions.refund(
      transactionCode,
      amount !== undefined ? { amount } : undefined
    );

    // Server returns 204 No Content -- return same shape as before
    return { success: true as const };
  } catch (error) {
    if (error instanceof APIError) {
      throw new Error(`SumUp refund failed: ${JSON.stringify(error.error)}`);
    }
    throw error;
  }
}
```

Key design decisions:
- **Singleton pattern**: `new SumUp()` instantiated once at module level, not per request. This avoids repeated allocations and enables future phases to import `sumup` directly.
- **Error normalization**: `APIError` is caught and rethrown as standard `Error` with the same message format as the old client. This ensures call sites that catch `Error` continue working. The message format matches what the old client produced (e.g., `"SumUp checkout creation failed: ..."`, `"SumUp refund failed: ..."`).
- **Currency cast**: `params.currency as "EUR"` -- the SDK expects a `Currency` union type. All call sites pass `"EUR"` as a string literal. The cast is safe because `"EUR"` is a valid member of the Currency type.
- **Return type assertions**: `checkout as { ... }` -- SDK types mark fields as optional (`id?: string`) because the OpenAPI spec is conservative. In practice, `create` and `get` always return these fields. The assertions preserve the existing return types that call sites depend on.
- **refundTransaction return**: Continues returning `{ success: true }` even though SDK returns `void`. No call site uses this value, but it maintains backward compatibility.
- **Export singleton**: `export { sumup }` exposes the raw SDK client for Phase 14+ (finance dashboard, payouts, etc.) which need `sumup.transactions.list()`, `sumup.payouts.list()`, etc.
